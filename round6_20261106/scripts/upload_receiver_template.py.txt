def receive_ram(spec):
    fd=os.open('/dev/tty',os.O_RDWR|os.O_NOCTTY)
    settings=termios.tcgetattr(fd)
    def emit(s):os.write(fd,(s+'\n').encode('ascii'))
    def exact(n):
        out=bytearray();deadline=time.monotonic()+20
        while len(out)<n:
            if not select.select([fd],[],[],max(0,deadline-time.monotonic()))[0]:
                raise TimeoutError('RAM packet input timeout')
            part=os.read(fd,n-len(out))
            if not part:raise EOFError('RAM tty input ended')
            out.extend(part)
        return bytes(out)
    data=bytearray()
    try:
        tty.setraw(fd,termios.TCSANOW)
        emit('\nRAM_READY_%s:%d:%s'%(spec['nonce'],spec['size'],spec['sha256']))
        seq=0
        while len(data)<spec['size']:
            header=exact(44)
            magic,index,size=struct.unpack('<4sII',header[:12])
            if magic!=b'KVPK' or index!=seq or not 1<=size<=32768 or len(data)+size>spec['size']:
                raise RuntimeError('RAM invalid packet header')
            packet=exact(size)
            if hashlib.sha256(packet).digest()!=header[12:]:
                raise RuntimeError('RAM packet SHA256 mismatch')
            data.extend(packet)
            emit('RAM_ACK_%s:%d:%d'%(spec['nonce'],seq,len(data)))
            seq+=1
        digest=hashlib.sha256(data).hexdigest()
        if digest!=spec['sha256']:raise RuntimeError('RAM whole payload SHA256 mismatch')
        emit('RAM_RECEIVED_%s:%d:%s'%(spec['nonce'],len(data),digest))
        return bytes(data)
    except Exception:
        # Discard only this uploader's remaining bytes before restoring the shell tty.
        end=time.monotonic()+8;quiet=time.monotonic()+3
        while time.monotonic()<end:
            if select.select([fd],[],[],max(0,min(end,quiet)-time.monotonic()))[0]:
                os.read(fd,65536);quiet=time.monotonic()+3
            else:break
        termios.tcflush(fd,termios.TCIFLUSH)
        raise
    finally:
        termios.tcsetattr(fd,termios.TCSANOW,settings)
        os.close(fd)
